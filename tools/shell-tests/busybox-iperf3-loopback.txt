# busybox-iperf3-loopback.txt — N70c first OSComp iperf3 loopback probe.
#
# Build image with:
#   TX_OSCOMP_RISCV_MUSL_DIR=/home/msp/learning/rustOS/testcase/riscv/musl \
#     cargo xtask image cpio --profile busybox --target rv64-qemu
#
# This is intentionally small: one one-off server in the background and
# one loopback TCP client. It is expected to become the main canary for
# the first iperf3 data-path fixes.

wait "/ # " within 30000
sleep 500

send "/bin/iperf3 -s -1 -p 5001 &\n"
expect "/ # " within 10000

sleep 1000
send "/bin/iperf3 -c 127.0.0.1 -p 5001 -t 1 -i 0\n"
expect "Connecting to host 127.0.0.1" within 10000
expect "sender" within 30000
expect "/ # " within 10000

quit
