# busybox-netns-procfs-caller-netns.txt - N72Q procfs caller-netns smoke.
#
# This verifies that normal read(2)/write(2) syscalls project /proc/net/*
# and /proc/sys/net/* through the caller's current network namespace.

wait "/ # " within 30000
sleep 500

send "/bin/netns-helper hold &\n"
expect "tx-netns-holder-ready" within 10000
expect "/ # " within 10000

send "m=N72Q; target=$!; echo ${m}_TARGET_$target\n"
expect "N72Q_TARGET_" within 5000
expect "/ # " within 5000

send "if /bin/busybox ip link add name docker0 type bridge; then echo ${m}_BRIDGE_CREATED; else echo ${m}_FAIL_BRIDGE_CREATE; fi\n"
expect "N72Q_BRIDGE_CREATED" within 15000
expect "/ # " within 10000

send "if /bin/busybox ip addr add 172.18.0.1/16 dev docker0; then echo ${m}_BRIDGE_ADDR; else echo ${m}_FAIL_BRIDGE_ADDR; fi\n"
expect "N72Q_BRIDGE_ADDR" within 15000
expect "/ # " within 10000

send "if /bin/busybox ip link set docker0 up; then echo ${m}_BRIDGE_UP; else echo ${m}_FAIL_BRIDGE_UP; fi\n"
expect "N72Q_BRIDGE_UP" within 15000
expect "/ # " within 10000

send "if /bin/busybox ip link add veth0 type veth peer name eth0; then echo ${m}_VETH_CREATED; else echo ${m}_FAIL_VETH_CREATE; fi\n"
expect "N72Q_VETH_CREATED" within 15000
expect "/ # " within 10000

send "if /bin/busybox ip link set dev veth0 master docker0 || /bin/netns-helper master veth0 docker0; then echo ${m}_VETH_MASTER; else echo ${m}_FAIL_VETH_MASTER; fi\n"
expect "N72Q_VETH_MASTER" within 15000
expect "/ # " within 10000

send "if /bin/busybox ip link set veth0 up; then echo ${m}_VETH_UP; else echo ${m}_FAIL_VETH_UP; fi\n"
expect "N72Q_VETH_UP" within 15000
expect "/ # " within 10000

send "if /bin/busybox ip link set eth0 netns $target || /bin/netns-helper netns eth0 $target; then echo ${m}_ETH_MOVED; else echo ${m}_FAIL_ETH_MOVE; fi\n"
expect "N72Q_ETH_MOVED" within 15000
expect "/ # " within 10000

send "if /bin/netns-helper exec $target /bin/busybox ip link set eth0 up; then echo ${m}_ETH_UP; else echo ${m}_FAIL_ETH_UP; fi\n"
expect "N72Q_ETH_UP" within 15000
expect "/ # " within 10000

send "if /bin/netns-helper exec $target /bin/busybox ip addr add 172.18.0.2/16 dev eth0; then echo ${m}_ETH_ADDR; else echo ${m}_FAIL_ETH_ADDR; fi\n"
expect "N72Q_ETH_ADDR" within 15000
expect "/ # " within 10000

send "if /bin/netns-helper exec $target /bin/busybox ip route add default via 172.18.0.1 dev eth0; then echo ${m}_ETH_DEFAULT_ROUTE; else echo ${m}_FAIL_ETH_DEFAULT_ROUTE; fi\n"
expect "N72Q_ETH_DEFAULT_ROUTE" within 15000
expect "/ # " within 10000

send "if echo 0 > /proc/sys/net/ipv4/ip_forward; then echo ${m}_HOST_FORWARD_ZERO; else echo ${m}_FAIL_HOST_FORWARD_ZERO; fi\n"
expect "N72Q_HOST_FORWARD_ZERO" within 10000
expect "/ # " within 10000

send "if /bin/netns-helper exec $target /bin/busybox sh -c 'echo 1 > /proc/sys/net/ipv4/ip_forward'; then echo ${m}_TARGET_FORWARD_ONE; else echo ${m}_FAIL_TARGET_FORWARD_ONE; fi\n"
expect "N72Q_TARGET_FORWARD_ONE" within 10000
expect "/ # " within 10000

send "cat /proc/sys/net/ipv4/ip_forward\n"
expect "0" within 5000
expect "/ # " within 5000

sleep 500
send "/bin/netns-helper exec $target /bin/busybox cat /proc/sys/net/ipv4/ip_forward\n"
expect "1" within 10000
expect "/ # " within 10000

send "if echo 'masquerade 172.18.0.0/16 docker0' > /proc/net/tx_nf_rules; then echo ${m}_HOST_NF_RULE; else echo ${m}_FAIL_HOST_NF_RULE; fi\n"
expect "N72Q_HOST_NF_RULE" within 10000
expect "/ # " within 10000

send "if /bin/netns-helper exec $target /bin/busybox sh -c \"echo 'filter forward ACCEPT out=eth0' > /proc/net/tx_nf_rules\"; then echo ${m}_TARGET_NF_RULE; else echo ${m}_FAIL_TARGET_NF_RULE; fi\n"
expect "N72Q_TARGET_NF_RULE" within 10000
expect "/ # " within 10000

send "if /bin/busybox grep docker0 /proc/net/dev >/dev/null && /bin/busybox grep docker0 /proc/net/route >/dev/null && /bin/busybox grep MASQUERADE /proc/net/tx_nf_rules >/dev/null && ! /bin/busybox grep eth0 /proc/net/tx_nf_rules >/dev/null; then echo ${m}_HOST_PROCFS_OK; else echo ${m}_FAIL_HOST_PROCFS; fi\n"
expect "N72Q_HOST_PROCFS_OK" within 10000
expect "/ # " within 10000

send "if /bin/netns-helper exec $target /bin/busybox sh -c '/bin/busybox grep eth0 /proc/net/dev >/dev/null && /bin/busybox grep eth0 /proc/net/route >/dev/null && /bin/busybox grep ACCEPT /proc/net/tx_nf_rules >/dev/null && ! /bin/busybox grep docker0 /proc/net/dev >/dev/null && ! /bin/busybox grep docker0 /proc/net/route >/dev/null && ! /bin/busybox grep docker0 /proc/net/tx_nf_rules >/dev/null'; then echo ${m}_TARGET_PROCFS_OK; else echo ${m}_FAIL_TARGET_PROCFS; fi\n"
expect "N72Q_TARGET_PROCFS_OK" within 10000
expect "/ # " within 10000

sleep 500
send "kill $target; echo ${m}_PROCFS_CALLER_NETNS_SUCCESS\n"
expect "N72Q_PROCFS_CALLER_NETNS_SUCCESS" within 10000
expect "/ # " within 10000

sleep 500
quit
