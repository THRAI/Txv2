# alpine-nft-iptables-smoke.txt - real Alpine RV64 nft/iptables user ABI probe.
#
# This script expects an alpine initramfs built with:
#   tools/images/fetch-alpine-rv64.sh
#   TX_ALPINE_ROOTFS=target/rootfs/alpine-rv64-qemu \
#     cargo xtask image cpio --profile alpine --target rv64-qemu
#
# Run with:
#   cargo xtask shell-test --target rv64-qemu --profile alpine \
#     --script tools/shell-tests/alpine-nft-iptables-smoke.txt

wait ":bootstrap-exec:ok" within 60000
wait "/ # " within 60000
sleep 500

send "echo alpine-shell-ok\n"
expect "alpine-shell-ok" within 10000
expect "/ # " within 10000

sleep 500
send "ls /sbin/nft /usr/sbin/nft /sbin/iptables /usr/sbin/iptables\n"
expect "nft" within 10000
expect "iptables" within 10000
expect "/ # " within 10000

sleep 500
send "/sbin/nft --version || /usr/sbin/nft --version\n"
expect "nftables" within 10000
expect "/ # " within 10000

sleep 500
send "/sbin/iptables --version || /usr/sbin/iptables --version\n"
expect "iptables" within 10000
expect "/ # " within 10000

sleep 500
send "/sbin/nft list ruleset || /usr/sbin/nft list ruleset\n"
expect "/ # " within 15000

sleep 500
quit
