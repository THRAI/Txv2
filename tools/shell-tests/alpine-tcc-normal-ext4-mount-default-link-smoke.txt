# alpine-tcc-normal-ext4-mount-default-link-smoke.txt - userland ext4 mount and TCC default link.
#
# Run with:
#   cargo xtask shell-test --target rv64-qemu --profile alpine \
#     --append-cmdline tx.mount.sdcard=0 \
#     --extra-rv64-ext4 target/images/alpine-tcc-dev-root-rv64-qemu.ext4 \
#     --script tools/shell-tests/alpine-tcc-normal-ext4-mount-default-link-smoke.txt

wait ":mount:bdevfs:ok" within 90000
wait "/ # " within 90000
sleep 500

send "echo normal-ext4-mount-tcc-start\n"
expect "normal-ext4-mount-tcc-start" within 10000
expect "/ # " within 10000

sleep 500
send "test -d /musl; echo boot-musl-dir-status:$?\n"
expect "boot-musl-dir-status:1" within 10000
expect "/ # " within 10000

sleep 500
send "mkdir -p /mnt/tcc; mount -t ext4 /dev/block/vda /mnt/tcc; echo normal-mount-status:$?\n"
expect "normal-mount-status:0" within 30000
expect "/ # " within 30000

sleep 500
send "test -x /mnt/tcc/usr/bin/tcc; echo mounted-tcc-status:$?\n"
expect "mounted-tcc-status:0" within 10000
expect "/ # " within 10000

sleep 500
send "mkdir -p /usr /lib; ln -sf /mnt/tcc/usr/include /usr/include; ln -sf /mnt/tcc/usr/lib/tcc /usr/lib/tcc; ln -sf /mnt/tcc/usr/lib/crt1.o /usr/lib/crt1.o; ln -sf /mnt/tcc/usr/lib/crti.o /usr/lib/crti.o; ln -sf /mnt/tcc/usr/lib/crtn.o /usr/lib/crtn.o; ln -sf /mnt/tcc/usr/lib/libc.a /usr/lib/libc.a; ln -sf /mnt/tcc/usr/lib/libc.so /usr/lib/libc.so; ln -sf /mnt/tcc/lib/ld-musl-riscv64.so.1 /lib/ld-musl-riscv64.so.1; echo tcc-compat-links:$?\n"
expect "tcc-compat-links:0" within 20000
expect "/ # " within 20000

sleep 500
send "printf '#include <stdio.h>\\nint main(void) { puts(\"tcc-normal-mount-linked\"); return 7; }\\n' > /tmp/tccmain.c\n"
expect "/ # " within 10000

sleep 500
send "/mnt/tcc/lib/ld-musl-riscv64.so.1 --library-path /mnt/tcc/lib:/mnt/tcc/usr/lib /mnt/tcc/usr/bin/tcc /tmp/tccmain.c -o /tmp/tccmain; echo default-link-status:$?\n"
expect "default-link-status:0" within 30000
expect "/ # " within 30000

sleep 500
send "/tmp/tccmain; echo default-run-status:$?\n"
expect "tcc-normal-mount-linked" within 20000
expect "default-run-status:7" within 20000
expect "/ # " within 20000

sleep 500
quit
