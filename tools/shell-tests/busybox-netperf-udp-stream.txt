# busybox-netperf-udp-stream.txt — N69c OSComp UDP_STREAM probe.
#
# Exercises netperf's IPv4 loopback UDP bulk datagram path with the exact
# OSComp musl arguments.

wait "/ # " within 30000
sleep 500

send "/bin/netserver -D -L 127.0.0.1 -p 12865 & server_pid=$!\n"
expect "/ # " within 10000
sleep 1500

send "echo \"====== netperf UDP_STREAM begin ======\"; if /bin/netperf -H 127.0.0.1 -p 12865 -t UDP_STREAM -l 1 -- -s 16k -S 16k -m 1k -M 1k; then echo \"N69C_UDP_STREAM_OK\"; else echo \"N69C_UDP_STREAM_FAIL\"; fi; kill -9 $server_pid\n"
expect "N69C_UDP_STREAM_OK" within 60000
expect "/ # " within 10000

sleep 500
quit
