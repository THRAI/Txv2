# alpine-sqlite-wal-smoke.txt - sqlite3 WAL-mode smoke with checkpoint.
wait "/ # " within 60000
sleep 500
send "rm -f /tmp/sqlite-wal-demo.db /tmp/sqlite-wal-demo.db-journal /tmp/sqlite-wal-demo.db-wal /tmp/sqlite-wal-demo.db-shm\n"
expect "/ # " within 10000
sleep 500
send "/usr/bin/sqlite3 -batch -noheader /tmp/sqlite-wal-demo.db \"PRAGMA journal_mode=WAL; PRAGMA synchronous=NORMAL; PRAGMA temp_store=MEMORY; CREATE TABLE wal_kv(k INTEGER PRIMARY KEY, v TEXT); INSERT INTO wal_kv VALUES(1,'one'),(2,'two'); UPDATE wal_kv SET v='two-updated' WHERE k=2; SELECT 'wal-count:' || count(*) FROM wal_kv; SELECT 'wal-row:' || v FROM wal_kv WHERE k=2; PRAGMA wal_checkpoint(FULL); PRAGMA integrity_check;\"; echo sqlite-wal-status-$?\n"
expect "wal" within 20000
expect "wal-count:2" within 20000
expect "wal-row:two-updated" within 20000
expect "ok" within 20000
expect "sqlite-wal-status-0" within 20000
sleep 500
send "/usr/bin/sqlite3 -batch -noheader /tmp/sqlite-wal-demo.db \"SELECT 'wal-reopen:' || v FROM wal_kv WHERE k=2; PRAGMA integrity_check;\"; echo sqlite-wal-reopen-status-$?\n"
expect "wal-reopen:two-updated" within 20000
expect "ok" within 20000
expect "sqlite-wal-reopen-status-0" within 20000
sleep 500
quit
