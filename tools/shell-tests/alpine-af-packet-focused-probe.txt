# alpine-af-packet-focused-probe.txt - AF_PACKET/link-layer socket probe.
#
# This keeps link-layer socket work network-scoped: a static helper opens
# AF_PACKET, resolves ifindex with SIOCGIFINDEX, binds sockaddr_ll, checks
# getsockname, and verifies nonblocking recvfrom reports EAGAIN.

wait ":bootstrap-exec:ok" within 60000
wait "/ # " within 60000
sleep 500

send "m=N73F; IP=/sbin/ip; [ -x $IP ] || IP=/usr/sbin/ip; echo ${m}_TOOLS_${IP}\n"
expect "N73F_TOOLS_" within 10000
expect "/ # " within 10000

sleep 500
send "if [ -x /bin/packet-probe ]; then echo ${m}_PACKET_PROBE_PRESENT; else echo ${m}_FAIL_NO_PACKET_PROBE; fi\n"
expect "N73F_PACKET_PROBE_PRESENT" within 10000
expect "/ # " within 10000

sleep 500
send "if /bin/packet-probe lo; then echo ${m}_PACKET_LO_OK; else echo ${m}_FAIL_PACKET_LO; fi\n"
expect "tx-packet-probe-ok" within 10000
expect "N73F_PACKET_LO_OK" within 10000
expect "/ # " within 10000

sleep 500
send "if $IP link add name docker0 type bridge; then echo ${m}_BRIDGE_CREATED; else echo ${m}_FAIL_BRIDGE_CREATE; fi\n"
expect "N73F_BRIDGE_CREATED" within 15000
expect "/ # " within 10000

sleep 500
send "if $IP link set docker0 up; then echo ${m}_BRIDGE_UP; else echo ${m}_FAIL_BRIDGE_UP; fi\n"
expect "N73F_BRIDGE_UP" within 15000
expect "/ # " within 10000

sleep 500
send "if /bin/packet-probe docker0; then echo ${m}_PACKET_DOCKER0_OK; else echo ${m}_FAIL_PACKET_DOCKER0; fi\n"
expect "tx-packet-probe-ok" within 10000
expect "N73F_PACKET_DOCKER0_OK" within 10000
expect "/ # " within 10000

sleep 500
send "if [ -x /bin/netns-helper ]; then echo ${m}_NETNS_HELPER_PRESENT; else echo ${m}_FAIL_NO_NETNS_HELPER; fi\n"
expect "N73F_NETNS_HELPER_PRESENT" within 10000
expect "/ # " within 10000

sleep 500
send "/bin/netns-helper hold &\n"
expect "tx-netns-holder-ready" within 10000
expect "/ # " within 10000

sleep 500
send "target=$!; echo ${m}_TARGET_${target}\n"
expect "N73F_TARGET_" within 10000
expect "/ # " within 10000

sleep 500
send "if $IP link add veth0 type veth peer name eth0; then echo ${m}_VETH_CREATED; else echo ${m}_FAIL_VETH_CREATE; fi\n"
expect "N73F_VETH_CREATED" within 15000
expect "/ # " within 10000

sleep 500
send "if $IP link set eth0 netns $target || /bin/netns-helper netns eth0 $target; then echo ${m}_ETH_MOVED; else echo ${m}_FAIL_ETH_MOVE; fi\n"
expect "N73F_ETH_MOVED" within 15000
expect "/ # " within 10000

sleep 500
send "if /bin/netns-helper exec $target $IP link set eth0 up; then echo ${m}_ETH_UP; else echo ${m}_FAIL_ETH_UP; fi\n"
expect "N73F_ETH_UP" within 15000
expect "/ # " within 10000

sleep 500
send "if /bin/netns-helper exec $target /bin/packet-probe eth0; then echo ${m}_PACKET_NETNS_ETH0_OK; else echo ${m}_FAIL_PACKET_NETNS_ETH0; fi\n"
expect "tx-packet-probe-ok" within 10000
expect "N73F_PACKET_NETNS_ETH0_OK" within 10000
expect "/ # " within 10000

sleep 500
send "kill $target; echo ${m}_PACKET_PROBE_SUCCESS\n"
expect "N73F_PACKET_PROBE_SUCCESS" within 10000
expect "/ # " within 10000

sleep 500
quit
