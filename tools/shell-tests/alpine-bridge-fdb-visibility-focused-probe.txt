# alpine-bridge-fdb-visibility-focused-probe.txt - Docker bridge/FDB surface.
#
# This probe keeps the next Docker-adjacent stage network-scoped. Real Alpine
# iproute2 drives detailed link, bridge link, bridge FDB, route, and neighbor
# visibility around a Docker-shaped bridge/veth setup.

wait ":bootstrap-exec:ok" within 60000
wait "/ # " within 60000
sleep 500

send "m=N73G; IP=/sbin/ip; [ -x $IP ] || IP=/usr/sbin/ip; BR=/sbin/bridge; [ -x $BR ] || BR=/usr/sbin/bridge; PING=/bin/ping; echo ${m}_TOOLS_${IP}_${BR}_${PING}\n"
expect "N73G_TOOLS_" within 10000
expect "/ # " within 10000

sleep 500
send "if [ -x $BR ]; then echo ${m}_BRIDGE_TOOL_PRESENT; else echo ${m}_FAIL_NO_BRIDGE_TOOL; fi\n"
expect "N73G_BRIDGE_TOOL_PRESENT" within 10000
expect "/ # " within 10000

sleep 500
send "if [ -x /bin/netns-helper ]; then echo ${m}_NETNS_HELPER_PRESENT; else echo ${m}_FAIL_NO_NETNS_HELPER; fi\n"
expect "N73G_NETNS_HELPER_PRESENT" within 10000
expect "/ # " within 10000

sleep 500
send "if [ -x $PING ]; then echo ${m}_PING_PRESENT; else echo ${m}_FAIL_NO_PING; fi\n"
expect "N73G_PING_PRESENT" within 10000
expect "/ # " within 10000

sleep 500
send "/bin/netns-helper hold &\n"
expect "tx-netns-holder-ready" within 10000
expect "/ # " within 10000

sleep 500
send "target=$!; echo ${m}_TARGET_${target}\n"
expect "N73G_TARGET_" within 10000
expect "/ # " within 10000

sleep 500
send "if $IP link add name docker0 type bridge; then echo ${m}_BRIDGE_CREATED; else echo ${m}_FAIL_BRIDGE_CREATE; fi\n"
expect "N73G_BRIDGE_CREATED" within 15000
expect "/ # " within 10000

sleep 500
send "if $IP addr add 172.17.0.1/16 dev docker0; then echo ${m}_BRIDGE_ADDR; else echo ${m}_FAIL_BRIDGE_ADDR; fi\n"
expect "N73G_BRIDGE_ADDR" within 15000
expect "/ # " within 10000

sleep 500
send "if $IP link set docker0 up; then echo ${m}_BRIDGE_UP; else echo ${m}_FAIL_BRIDGE_UP; fi\n"
expect "N73G_BRIDGE_UP" within 15000
expect "/ # " within 10000

sleep 500
send "if $IP link add veth0 type veth peer name eth0; then echo ${m}_VETH_CREATED; else echo ${m}_FAIL_VETH_CREATE; fi\n"
expect "N73G_VETH_CREATED" within 15000
expect "/ # " within 10000

sleep 500
send "if $IP link set veth0 master docker0 || /bin/netns-helper master veth0 docker0; then echo ${m}_VETH_MASTER; else echo ${m}_FAIL_VETH_MASTER; fi\n"
expect "N73G_VETH_MASTER" within 15000
expect "/ # " within 10000

sleep 500
send "if $IP link set veth0 up; then echo ${m}_VETH_UP; else echo ${m}_FAIL_VETH_UP; fi\n"
expect "N73G_VETH_UP" within 15000
expect "/ # " within 10000

sleep 500
send "if $IP link set eth0 netns $target || /bin/netns-helper netns eth0 $target; then echo ${m}_ETH_MOVED; else echo ${m}_FAIL_ETH_MOVE; fi\n"
expect "N73G_ETH_MOVED" within 15000
expect "/ # " within 10000

sleep 500
send "if /bin/netns-helper exec $target $IP link set eth0 up; then echo ${m}_ETH_UP; else echo ${m}_FAIL_ETH_UP; fi\n"
expect "N73G_ETH_UP" within 15000
expect "/ # " within 10000

sleep 500
send "if /bin/netns-helper exec $target $IP addr add 172.17.0.2/16 dev eth0; then echo ${m}_ETH_ADDR; else echo ${m}_FAIL_ETH_ADDR; fi\n"
expect "N73G_ETH_ADDR" within 15000
expect "/ # " within 10000

sleep 500
send "if $IP -d link show dev docker0; then echo ${m}_IP_DETAIL_DOCKER0_OK; else echo ${m}_FAIL_IP_DETAIL_DOCKER0; fi\n"
expect "bridge" within 10000
expect "N73G_IP_DETAIL_DOCKER0_OK" within 10000
expect "/ # " within 10000

sleep 500
send "if $IP -d link show dev veth0; then echo ${m}_IP_DETAIL_VETH0_OK; else echo ${m}_FAIL_IP_DETAIL_VETH0; fi\n"
expect "veth" within 10000
expect "N73G_IP_DETAIL_VETH0_OK" within 10000
expect "/ # " within 10000

sleep 500
send "if $BR link show; then echo ${m}_BRIDGE_LINK_OK; else echo ${m}_FAIL_BRIDGE_LINK; fi\n"
expect "veth0" within 10000
expect "docker0" within 10000
expect "N73G_BRIDGE_LINK_OK" within 10000
expect "/ # " within 10000

sleep 500
send "if $BR fdb show br docker0; then echo ${m}_BRIDGE_FDB_OK; else echo ${m}_FAIL_BRIDGE_FDB; fi\n"
expect "N73G_BRIDGE_FDB_OK" within 10000
expect "/ # " within 10000

sleep 500
send "if /bin/netns-helper exec $target $PING -c 1 172.17.0.1; then echo ${m}_PING_BRIDGE_OK; else echo ${m}_FAIL_PING_BRIDGE; fi\n"
expect "N73G_PING_BRIDGE_OK" within 20000
expect "/ # " within 10000

sleep 500
send "if $IP neigh show dev docker0; then echo ${m}_NEIGH_DOCKER0_OK; else echo ${m}_FAIL_NEIGH_DOCKER0; fi\n"
expect "172.17.0.2" within 10000
expect "N73G_NEIGH_DOCKER0_OK" within 10000
expect "/ # " within 10000

sleep 500
send "if $BR fdb show br docker0; then echo ${m}_BRIDGE_FDB_AFTER_PING_OK; else echo ${m}_FAIL_BRIDGE_FDB_AFTER_PING; fi\n"
expect "N73G_BRIDGE_FDB_AFTER_PING_OK" within 10000
expect "/ # " within 10000

sleep 500
send "kill $target; echo ${m}_BRIDGE_VISIBILITY_SUCCESS\n"
expect "N73G_BRIDGE_VISIBILITY_SUCCESS" within 10000
expect "/ # " within 10000

sleep 500
quit
