# busybox-iperf3-reverse-udp-oneshot.txt — isolate iperf3 UDP reverse mode.

wait "/ # " within 30000
sleep 500

send "/bin/iperf3 -s -1 -p 5001 &\n"
expect "/ # " within 10000

sleep 1000
send "ok='N70_IPERF3_REVERSE_UDP_ONESHOT_suc''cess'; fail='N70_IPERF3_REVERSE_UDP_ONESHOT_fa''il'; if /bin/iperf3 -c 127.0.0.1 -p 5001 -t 2 -i 0 -u -R -b 1000G; then echo \"$ok\"; else echo \"$fail\"; fi\n"
expect "N70_IPERF3_REVERSE_UDP_ONESHOT_success" within 60000
expect "/ # " within 10000

sleep 500
quit
