# busybox-netperf-oscomp-suite.txt — N69c OSComp netperf-musl suite.
#
# Mirrors /home/msp/learning/rustOS/testcase/riscv/musl/netperf_testcode.sh
# with explicit success sentinels for shell-test.

wait "/ # " within 30000
sleep 500

send "echo \"#### OS COMP TEST GROUP START netperf-musl ####\"\n"
expect "#### OS COMP TEST GROUP START netperf-musl ####" within 10000
expect "/ # " within 10000

sleep 500
send "/bin/netserver -D -L 127.0.0.1 -p 12865 & server_pid=$!\n"
expect "/ # " within 10000
sleep 1500

send "echo \"====== netperf UDP_STREAM begin ======\"; if /bin/netperf -H 127.0.0.1 -p 12865 -t UDP_STREAM -l 1 -- -s 16k -S 16k -m 1k -M 1k; then echo \"====== netperf UDP_STREAM end: success ======\"; else echo \"====== netperf UDP_STREAM end: fail ======\"; fi\n"
expect "====== netperf UDP_STREAM end: success ======" within 60000
expect "/ # " within 10000

sleep 500
send "echo \"====== netperf TCP_STREAM begin ======\"; if /bin/netperf -H 127.0.0.1 -p 12865 -t TCP_STREAM -l 1 -- -s 16k -S 16k -m 1k -M 1k; then echo \"====== netperf TCP_STREAM end: success ======\"; else echo \"====== netperf TCP_STREAM end: fail ======\"; fi\n"
expect "====== netperf TCP_STREAM end: success ======" within 60000
expect "/ # " within 10000

sleep 500
send "echo \"====== netperf UDP_RR begin ======\"; if /bin/netperf -H 127.0.0.1 -p 12865 -t UDP_RR -l 1 -- -s 16k -S 16k -m 1k -M 1k -r 64,64 -R 1; then echo \"====== netperf UDP_RR end: success ======\"; else echo \"====== netperf UDP_RR end: fail ======\"; fi\n"
expect "====== netperf UDP_RR end: success ======" within 60000
expect "/ # " within 10000

sleep 500
send "echo \"====== netperf TCP_RR begin ======\"; if /bin/netperf -H 127.0.0.1 -p 12865 -t TCP_RR -l 1 -- -s 16k -S 16k -m 1k -M 1k -r 64,64 -R 1; then echo \"====== netperf TCP_RR end: success ======\"; else echo \"====== netperf TCP_RR end: fail ======\"; fi\n"
expect "====== netperf TCP_RR end: success ======" within 60000
expect "/ # " within 10000

sleep 500
send "echo \"====== netperf TCP_CRR begin ======\"; if /bin/netperf -H 127.0.0.1 -p 12865 -t TCP_CRR -l 1 -- -s 16k -S 16k -m 1k -M 1k -r 64,64 -R 1; then echo \"====== netperf TCP_CRR end: success ======\"; else echo \"====== netperf TCP_CRR end: fail ======\"; fi\n"
expect "====== netperf TCP_CRR end: success ======" within 60000
expect "/ # " within 10000

sleep 500
send "kill -9 $server_pid; echo \"#### OS COMP TEST GROUP END netperf-musl ####\"\n"
expect "#### OS COMP TEST GROUP END netperf-musl ####" within 10000
expect "/ # " within 10000

sleep 500
quit
