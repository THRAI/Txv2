# busybox-udp-loopback.txt — N68 BusyBox shell UDP loopback smoke.
#
# Run with:
#   cargo xtask image cpio --profile busybox --target rv64-qemu
#   cargo xtask shell-test --target rv64-qemu --script tools/shell-tests/busybox-udp-loopback.txt
#
# The shell launches a tiny freestanding static test binary that creates a UDP
# loopback socket, binds 127.0.0.1:2325, sends one datagram to itself, receives
# it back, and returns to the prompt.

wait "/ # " within 30000
sleep 500

send "/bin/udp-loopback-smoke\n"
expect "tx-n68-udp-ok" within 10000
expect "/ # " within 10000

quit
