# alpine-nft-oifname-masquerade.txt - real nft interface-match NAT rule.
#
# Docker-style NAT rules commonly include an output-interface predicate before
# MASQUERADE. This keeps the real Alpine nft frontend wired through
# NETLINK_NETFILTER rather than the local /bin/nft-probe helper.

wait ":bootstrap-exec:ok" within 60000
wait "/ # " within 60000
sleep 500

send "NFT=/sbin/nft; [ -x $NFT ] || NFT=/usr/sbin/nft; $NFT --version\n"
expect "nftables" within 10000
expect "/ # " within 10000

sleep 500
send "$NFT add table ip nat\n"
expect "/ # " within 15000

sleep 500
send "$NFT add chain ip nat postrouting '{ type nat hook postrouting priority 100; }'\n"
expect "/ # " within 15000

sleep 500
send "$NFT add rule ip nat postrouting oifname \"docker0\" masquerade\n"
expect "/ # " within 15000

sleep 500
send "$NFT list ruleset\n"
expect "table ip nat" within 15000
expect "chain postrouting" within 15000
expect "oifname" within 15000
expect "docker0" within 15000
expect "masquerade" within 15000
expect "/ # " within 15000

sleep 500
send "$NFT delete table ip nat\n"
expect "/ # " within 15000

sleep 500
quit
