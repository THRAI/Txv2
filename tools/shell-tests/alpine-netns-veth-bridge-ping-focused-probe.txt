# alpine-netns-veth-bridge-ping-focused-probe.txt - Alpine netns data plane.
#
# This extends the Alpine netns control probe into packet-carrying checks:
# - namespace A eth0 pings host docker0 at 172.17.0.1.
# - namespace A eth0 pings namespace B eth0 through docker0.
# - target namespace route and neighbor state are visible through iproute2.

wait ":bootstrap-exec:ok" within 60000
wait "/ # " within 60000
sleep 500

send "m=N73D; IP=/sbin/ip; PING=/bin/ping; echo ${m}_TOOLS_${IP}_${PING}\n"
expect "N73D_TOOLS_" within 10000
expect "/ # " within 10000

sleep 500
send "if [ -x /bin/netns-helper ]; then echo ${m}_NETNS_HELPER_PRESENT; else echo ${m}_FAIL_NO_NETNS_HELPER; fi\n"
expect "N73D_NETNS_HELPER_PRESENT" within 10000
expect "/ # " within 10000

sleep 500
send "if [ -x $PING ]; then echo ${m}_PING_PRESENT; else echo ${m}_FAIL_NO_PING; fi\n"
expect "N73D_PING_PRESENT" within 10000
expect "/ # " within 10000

sleep 500
send "/bin/netns-helper hold &\n"
expect "tx-netns-holder-ready" within 10000
expect "/ # " within 10000

sleep 500
send "target_a=$!; echo ${m}_TARGET_A_${target_a}\n"
expect "N73D_TARGET_A_" within 10000
expect "/ # " within 10000

sleep 500
send "/bin/netns-helper hold &\n"
expect "tx-netns-holder-ready" within 10000
expect "/ # " within 10000

sleep 500
send "target_b=$!; echo ${m}_TARGET_B_${target_b}\n"
expect "N73D_TARGET_B_" within 10000
expect "/ # " within 10000

sleep 500
send "if $IP link add name docker0 type bridge; then echo ${m}_BRIDGE_CREATED; else echo ${m}_FAIL_BRIDGE_CREATE; fi\n"
expect "N73D_BRIDGE_CREATED" within 15000
expect "/ # " within 10000

sleep 500
send "if $IP addr add 172.17.0.1/16 dev docker0; then echo ${m}_BRIDGE_ADDR; else echo ${m}_FAIL_BRIDGE_ADDR; fi\n"
expect "N73D_BRIDGE_ADDR" within 15000
expect "/ # " within 10000

sleep 500
send "if $IP link set docker0 up; then echo ${m}_BRIDGE_UP; else echo ${m}_FAIL_BRIDGE_UP; fi\n"
expect "N73D_BRIDGE_UP" within 15000
expect "/ # " within 10000

sleep 500
send "if $IP link add veth0 type veth peer name eth0; then echo ${m}_VETH_A_CREATED; else echo ${m}_FAIL_VETH_A_CREATE; fi\n"
expect "N73D_VETH_A_CREATED" within 15000
expect "/ # " within 10000

sleep 500
send "if $IP link set veth0 master docker0 || /bin/netns-helper master veth0 docker0; then echo ${m}_VETH_A_MASTER; else echo ${m}_FAIL_VETH_A_MASTER; fi\n"
expect "N73D_VETH_A_MASTER" within 15000
expect "/ # " within 10000

sleep 500
send "if $IP link set veth0 up; then echo ${m}_VETH_A_UP; else echo ${m}_FAIL_VETH_A_UP; fi\n"
expect "N73D_VETH_A_UP" within 15000
expect "/ # " within 10000

sleep 500
send "if $IP link set eth0 netns $target_a || /bin/netns-helper netns eth0 $target_a; then echo ${m}_ETH_A_MOVED; else echo ${m}_FAIL_ETH_A_MOVE; fi\n"
expect "N73D_ETH_A_MOVED" within 15000
expect "/ # " within 10000

sleep 500
send "if /bin/netns-helper exec $target_a $IP link set eth0 up; then echo ${m}_ETH_A_UP; else echo ${m}_FAIL_ETH_A_UP; fi\n"
expect "N73D_ETH_A_UP" within 15000
expect "/ # " within 10000

sleep 500
send "if /bin/netns-helper exec $target_a $IP addr add 172.17.0.2/16 dev eth0; then echo ${m}_ETH_A_ADDR; else echo ${m}_FAIL_ETH_A_ADDR; fi\n"
expect "N73D_ETH_A_ADDR" within 15000
expect "/ # " within 10000

sleep 500
send "if /bin/netns-helper exec $target_a $IP route add default via 172.17.0.1 dev eth0; then echo ${m}_ETH_A_DEFAULT_ROUTE; else echo ${m}_FAIL_ETH_A_DEFAULT_ROUTE; fi\n"
expect "N73D_ETH_A_DEFAULT_ROUTE" within 15000
expect "/ # " within 10000

sleep 500
send "if $IP link add veth1 type veth peer name eth0; then echo ${m}_VETH_B_CREATED; else echo ${m}_FAIL_VETH_B_CREATE; fi\n"
expect "N73D_VETH_B_CREATED" within 15000
expect "/ # " within 10000

sleep 500
send "if $IP link set veth1 master docker0 || /bin/netns-helper master veth1 docker0; then echo ${m}_VETH_B_MASTER; else echo ${m}_FAIL_VETH_B_MASTER; fi\n"
expect "N73D_VETH_B_MASTER" within 15000
expect "/ # " within 10000

sleep 500
send "if $IP link set veth1 up; then echo ${m}_VETH_B_UP; else echo ${m}_FAIL_VETH_B_UP; fi\n"
expect "N73D_VETH_B_UP" within 15000
expect "/ # " within 10000

sleep 500
send "if $IP link set eth0 netns $target_b || /bin/netns-helper netns eth0 $target_b; then echo ${m}_ETH_B_MOVED; else echo ${m}_FAIL_ETH_B_MOVE; fi\n"
expect "N73D_ETH_B_MOVED" within 15000
expect "/ # " within 10000

sleep 500
send "if /bin/netns-helper exec $target_b $IP link set eth0 up; then echo ${m}_ETH_B_UP; else echo ${m}_FAIL_ETH_B_UP; fi\n"
expect "N73D_ETH_B_UP" within 15000
expect "/ # " within 10000

sleep 500
send "if /bin/netns-helper exec $target_b $IP addr add 172.17.0.3/16 dev eth0; then echo ${m}_ETH_B_ADDR; else echo ${m}_FAIL_ETH_B_ADDR; fi\n"
expect "N73D_ETH_B_ADDR" within 15000
expect "/ # " within 10000

sleep 500
send "if /bin/netns-helper exec $target_b $IP route add default via 172.17.0.1 dev eth0; then echo ${m}_ETH_B_DEFAULT_ROUTE; else echo ${m}_FAIL_ETH_B_DEFAULT_ROUTE; fi\n"
expect "N73D_ETH_B_DEFAULT_ROUTE" within 15000
expect "/ # " within 10000

sleep 500
send "/bin/netns-helper exec $target_a $IP route\n"
expect "172.17.0.0" within 10000
expect "default" within 10000
expect "eth0" within 10000
expect "/ # " within 10000

sleep 500
send "if /bin/netns-helper exec $target_a $PING -c 1 -W 5 172.17.0.1; then echo ${m}_PING_HOST_OK; else echo ${m}_FAIL_PING_HOST; fi\n"
expect "N73D_PING_HOST_OK" within 30000
expect "/ # " within 10000

sleep 500
send "if /bin/netns-helper exec $target_a $PING -c 1 -W 5 172.17.0.3; then echo ${m}_PING_PEER_OK; else echo ${m}_FAIL_PING_PEER; fi\n"
expect "N73D_PING_PEER_OK" within 30000
expect "/ # " within 10000

sleep 500
send "/bin/netns-helper exec $target_a $IP neigh\n"
expect "172.17.0.1" within 10000
expect "172.17.0.3" within 10000
expect "/ # " within 10000

sleep 500
send "kill $target_a $target_b; echo ${m}_BRIDGE_PING_SUCCESS\n"
expect "N73D_BRIDGE_PING_SUCCESS" within 10000
expect "/ # " within 10000

sleep 500
quit
