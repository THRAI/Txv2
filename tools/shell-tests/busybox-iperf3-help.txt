# busybox-iperf3-help.txt — N70a OSComp iperf3 startup smoke.
#
# Build image with:
#   TX_OSCOMP_RISCV_MUSL_DIR=/home/msp/learning/rustOS/testcase/riscv/musl \
#     cargo xtask image cpio --profile busybox --target rv64-qemu

# This only proves the external static musl iperf3 binary is present,

expect "/ # " within 10000

quit
