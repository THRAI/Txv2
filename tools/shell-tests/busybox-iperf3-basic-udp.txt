# busybox-iperf3-basic-udp.txt — OSComp iperf BASIC_UDP probe.
#
# Small repro for the first official iperf_testcode.sh case:
#   iperf3 -s -p 5001 -D
#   iperf3 -c 127.0.0.1 -p 5001 -t 2 -i 0 -u -b 1000G

wait "/ # " within 30000
sleep 500

send "/bin/iperf3 -s -p 5001 -D\n"
expect "/ # " within 10000

sleep 1500
send "ok='N70_IPERF3_BASIC_UDP_suc''cess'; fail='N70_IPERF3_BASIC_UDP_fa''il'; if /bin/iperf3 -c 127.0.0.1 -p 5001 -t 2 -i 0 -u -b 1000G; then echo \"$ok\"; else echo \"$fail\"; fi\n"
expect "N70_IPERF3_BASIC_UDP_success" within 45000
expect "/ # " within 10000

sleep 500
quit
